@torch.no_grad()
def set_regulatory_bias(self, bias: torch.Tensor, *, clip: float) -> None:
    if bias.shape != self.expert_bias.shape:

    if not torch.isfinite(bias).all():

    bounded = bias.to(self.expert_bias).sub(bias.to(self.expert_bias).mean())

def update_target_ema(self, decay: float) -> None:

Create a parameter-free `nn.Module` with buffers shaped
`[n_layers, n_experts]`:

```python
@dataclass(frozen=True)
class LearningProgressConfig:
    n_layers: int
    n_experts: int
    mode: str = "shadow"
    expected_decay: float = 0.95
    eligibility_decay: float = 0.95
    bias_clip: float = 0.25
    bias_gain: float = 0.05
    jepa_weight: float = 0.25
    forgetting_weight: float = 0.5
    load_weight: float = 0.1


class LearningProgressController(nn.Module):
    def __init__(self, config: LearningProgressConfig) -> None:
        super().__init__()
        if config.mode not in {"disabled", "control", "shadow", "enforce"}:
            raise ValueError("invalid learning-progress mode")
        self.config = config
        shape = (config.n_layers, config.n_experts)
        for name in (
            "eligibility", "expected_progress", "delta", "last_lm",
            "last_jepa", "credit", "router_bias",
        ):
            self.register_buffer(name, torch.zeros(shape, dtype=torch.float32))
        self.register_buffer("probe_count", torch.zeros((), dtype=torch.long))

    @torch.no_grad()
    def observe_routing(self, routing_mass: torch.Tensor) -> None:
        mass = routing_mass.to(self.eligibility)
        if mass.shape != self.eligibility.shape or not torch.isfinite(mass).all():
            raise ValueError("invalid routing mass")
        decay = self.config.eligibility_decay
        self.eligibility.mul_(decay).add_(mass, alpha=1.0 - decay)

    @torch.no_grad()
    def update_probe(
        self,
        *,
        lm_by_expert: torch.Tensor,
        jepa_by_expert: torch.Tensor,
        forgetting_by_expert: torch.Tensor,
        load_imbalance: torch.Tensor,
    ) -> dict[str, torch.Tensor]:
        lm = lm_by_expert.to(self.last_lm)
        jepa = jepa_by_expert.to(self.last_jepa)
        if self.probe_count.item() == 0:
            self.last_lm.copy_(lm)
            self.last_jepa.copy_(jepa)
            self.probe_count.add_(1)
            return {"delta": self.delta.clone(), "credit": self.credit.clone()}
        progress = (
            self.last_lm - lm
            + self.config.jepa_weight * (self.last_jepa - jepa)
            - self.config.forgetting_weight * forgetting_by_expert.to(lm)
            - self.config.load_weight * load_imbalance.to(lm)
        )
        self.delta.copy_(progress - self.expected_progress)
        decay = self.config.expected_decay
        self.expected_progress.mul_(decay).add_(progress, alpha=1.0 - decay)
        self.credit.copy_(self.delta * self.eligibility)
        self.last_lm.copy_(lm)
        self.last_jepa.copy_(jepa)
        self.router_bias.copy_(self.proposed_bias())
        self.probe_count.add_(1)
        return {"delta": self.delta.clone(), "credit": self.credit.clone()}

    @torch.no_grad()
    def proposed_bias(self) -> torch.Tensor:
        bias = self.credit * self.config.bias_gain
        bias = bias - bias.mean(dim=-1, keepdim=True)
        bias = bias.clamp(-self.config.bias_clip, self.config.bias_clip)
        return bias - bias.mean(dim=-1, keepdim=True)

    @torch.no_grad()
    def applied_bias(self) -> torch.Tensor:
        if self.config.mode != "enforce":
            return torch.zeros_like(self.router_bias)
        return self.router_bias.clone()
```
